  // Render input type selection

  document.querySelectorAll('.input-type-btn').forEach(btn => {

    btn.addEventListener('click', () => {

      const newType = btn.dataset.type;

      if (newType === state.inputType) return;

      if (state.selected.size > 0 || state.file) {

        if (!confirm('Switching will clear your uploaded file and selected presets. Continue?')) return;

      }

      state.inputType = newType;

      state.selected.clear();

      state.file = null;

      state.fileDataUrl = null;

      state.activeCategory = 'All';

      document.querySelectorAll('.input-type-btn').forEach(b => b.classList.toggle('active', b.dataset.type === newType));

      dropzoneMain.textContent = HINTS[newType].main;

      dropzoneHint.textContent = HINTS[newType].hint;

      step2Title.textContent = TITLES[newType];

      preview.innerHTML = '<div class="preview-empty">preview</div>';

      renderChips();

      renderPresets();

      updateUI();

    });

  });


  function renderChips() {

    const cats = PRESETS[state.inputType].categories;

    chips.innerHTML = cats.map(cat => {

      const active = cat === state.activeCategory ? 'active' : '';

      const isVideo = cat === 'Video';

      const playIcon = isVideo ? '<svg width="9" height="9" viewBox="0 0 24 24" fill="currentColor"><polygon points="6 4 20 12 6 20"/></svg>' : '';

      return `<button class="chip ${active}" data-cat="${cat}">${playIcon}${cat}</button>`;

    }).join('');

    chips.querySelectorAll('.chip').forEach(chip => {

      chip.addEventListener('click', () => {

        state.activeCategory = chip.dataset.cat;

        renderChips();

        renderPresets();

      });

    });

  }


  function renderPresets() {

    const items = PRESETS[state.inputType].items;

    const filtered = state.activeCategory === 'All' ? items : items.filter(i => i.cat === state.activeCategory);

    presetGrid.innerHTML = filtered.map(item => {

      const id = `${state.inputType}-${item.name}`;

      const selected = state.selected.has(id) ? 'selected' : '';

      const border = item.border ? 'border: 0.5px solid var(--border-tertiary);' : '';

      const playOverlay = item.video ? `<div class="play-overlay"><svg width="14" height="14" viewBox="0 0 24 24" fill="white"><polygon points="6 4 20 12 6 20"/></svg></div>` : '';

      return `

        <div class="preset-card ${selected}" data-id="${id}">

          <div class="preset-thumb" style="background:${item.bg};${border}">${playOverlay}</div>

          <div class="check"><svg width="9" height="9" viewBox="0 0 24 24" fill="none" stroke="white" stroke-width="3"><polyline points="20 6 9 17 4 12"/></svg></div>

          <div class="preset-name">${item.name}</div>

        </div>

      `;

    }).join('');

    presetGrid.querySelectorAll('.preset-card').forEach(card => {

      card.addEventListener('click', () => {

        const id = card.dataset.id;

        if (state.selected.has(id)) state.selected.delete(id);

        else state.selected.add(id);

        renderPresets();

        updateUI();

      });

    });

  }


  function updateUI() {

    selectedCount.textContent = `${state.selected.size} selected`;

    const hasKey = !!state.apiKey;

    const hasFile = !!state.file;

    const hasSelected = state.selected.size > 0;

    if (hasKey) {

      footerStatus.textContent = 'fal.ai key connected';

      footerStatus.classList.remove('warn');

    } else {

      footerStatus.textContent = 'Connect your fal.ai key →';

      footerStatus.classList.add('warn');

    }

    generateBtn.disabled = !(hasKey && hasFile && hasSelected) || state.generating;

    generateBtn.textContent = hasSelected ? `Generate ${state.selected.size} mockup${state.selected.size === 1 ? '' : 's'} →` : 'Generate →';

  }


  // Dropzone

  dropzone.addEventListener('click', () => fileInput.click());

  dropzone.addEventListener('dragover', e => { e.preventDefault(); dropzone.classList.add('dragover'); });

  dropzone.addEventListener('dragleave', () => dropzone.classList.remove('dragover'));

  dropzone.addEventListener('drop', e => {

    e.preventDefault();

    dropzone.classList.remove('dragover');

    if (e.dataTransfer.files.length) handleFile(e.dataTransfer.files[0]);

  });

  fileInput.addEventListener('change', e => {

    if (e.target.files.length) handleFile(e.target.files[0]);

  });


  function handleFile(file) {

    state.file = file;

    const reader = new FileReader();

    reader.onload = () => {

      state.fileDataUrl = reader.result;

      preview.innerHTML = `<img src="${reader.result}" alt="upload">`;

      updateUI();

    };

    reader.readAsDataURL(file);

  }


  // Settings modal

  settingsBtn.addEventListener('click', () => {

    keyInput.value = state.apiKey;

    modal.classList.add('visible');

    setTimeout(() => keyInput.focus(), 50);

  });

  $('modal-cancel').addEventListener('click', () => modal.classList.remove('visible'));

  $('modal-save').addEventListener('click', () => {

    state.apiKey = keyInput.value.trim();

    if (state.apiKey) localStorage.setItem('mockstack-fal-key', state.apiKey);

    else localStorage.removeItem('mockstack-fal-key');

    modal.classList.remove('visible');

    updateUI();

  });

  modal.addEventListener('click', e => {

    if (e.target === modal) modal.classList.remove('visible');

  });


  // Generate (fake)

  generateBtn.addEventListener('click', () => {

    if (state.generating) return;

    state.generating = true;

    updateUI();


    const items = PRESETS[state.inputType].items;

    const selectedItems = items.filter(i => state.selected.has(`${state.inputType}-${i.name}`));


    resultsPanel.classList.add('visible');

    resultsGrid.innerHTML = selectedItems.map((item, i) => `

      <div class="result-card" data-idx="${i}">

        <div class="thumb queue-tile">Queued</div>

        <div class="result-name">${item.name}</div>

      </div>

    `).join('');

    resultsPanel.scrollIntoView({ behavior: 'smooth', block: 'start' });


    // Simulate generation: each tile goes Queued → Generating → Done

    selectedItems.forEach((item, i) => {

      setTimeout(() => {

        const card = resultsGrid.querySelector(`[data-idx="${i}"] .thumb`);

        if (card) {

          card.className = 'thumb spinner-tile';

          card.innerHTML = '<div class="spinner"></div><span style="font-size:10px;color:var(--text-tertiary);">Generating</span>';

        }

      }, 400 + i * 600);


      setTimeout(() => {

        const card = resultsGrid.querySelector(`[data-idx="${i}"] .thumb`);

        if (card) {

          const border = item.border ? 'border: 0.5px solid var(--border-tertiary);' : '';

          card.className = 'thumb';

          card.style.background = item.bg;

          if (item.border) card.style.border = '0.5px solid var(--border-tertiary)';

          card.innerHTML = '';

          // Hint at composited result with a soft inner element

          const inner = document.createElement('div');

          inner.style.cssText = 'position:absolute;inset:18px;background:rgba(255,255,255,0.6);border-radius:4px;';

          card.appendChild(inner);

        }

        if (i === selectedItems.length - 1) {

          state.generating = false;

          updateUI();

        }

      }, 1800 + i * 800);

    });

  });


  // Initial render

  renderChips();

  renderPresets();

  updateUI();

</script>


</body>

</html>
